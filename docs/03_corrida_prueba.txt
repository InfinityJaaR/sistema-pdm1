-- CONSULTAS DE VERIFICACIÓN FINAL
-- ============================================================

-- Resumen de vehículos por estado
SELECT ESTADO_VEHICULO, COUNT(*) AS TOTAL
FROM VEHICULO
GROUP BY ESTADO_VEHICULO;

-- Capacidades de secciones usadas
SELECT S.ID_SECCION, S.NIVEL, B.NOMBRE_BODEGA,
       S.CAPACIDAD_ACTUAL, S.CAPACIDAD_MAXIMA
FROM SECCION S
JOIN BODEGA B ON B.ID_BODEGA = S.ID_BODEGA
WHERE S.CAPACIDAD_ACTUAL > 0;

-- Importadores con cantidad de vehículos
SELECT I.NOMBRE_IMPORTADOR, I.APELLIDO_IMPORTADOR,
       COUNT(V.ID_VEHICULO) AS VEHICULOS
FROM IMPORTADOR I
JOIN IMPORTACION IM ON IM.ID_IMPORTADOR = I.ID_IMPORTADOR
JOIN VEHICULO V ON V.ID_IMPORTACION = IM.ID_IMPORTACION
GROUP BY I.ID_IMPORTADOR;

-- Vehículos en reparación activa
SELECT V.VIN, V.ESTADO_VEHICULO, T.NOMBRE_TALLER,
       R.FECHA_INICIO, R.FECHA_FIN
FROM REPARACION R
JOIN VEHICULO V ON V.ID_VEHICULO = R.ID_VEHICULO
JOIN TALLER T ON T.ID_TALLER = R.ID_TALLER
WHERE R.FECHA_FIN IS NULL;
