-- ============================================================
-- CONSULTAS DE VERIFICACIÓN FINAL
-- ============================================================

-- Resumen de vehículos por estado
-- Esperado: 3 'en reparacion', 2 'vendido'
SELECT ESTADO_VEHICULO, COUNT(*) AS TOTAL
FROM VEHICULO
GROUP BY ESTADO_VEHICULO;

-- Capacidades de secciones usadas
SELECT S.ID_SECCION, S.NIVEL, B.NOMBRE_BODEGA,
       S.CAPACIDAD_ACTUAL, S.CAPACIDAD_MAXIMA
FROM SECCION S
JOIN BODEGA B ON B.ID_BODEGA = S.ID_BODEGA
WHERE S.CAPACIDAD_ACTUAL > 0;

-- Importadores con cantidad de vehículos
SELECT I.NOMBRE_IMPORTADOR, I.APELLIDO_IMPORTADOR,
       COUNT(V.ID_VEHICULO) AS VEHICULOS
FROM IMPORTADOR I
JOIN IMPORTACION IM ON IM.ID_IMPORTADOR = I.ID_IMPORTADOR
JOIN VEHICULO V ON V.ID_IMPORTACION = IM.ID_IMPORTACION
GROUP BY I.ID_IMPORTADOR;

-- Vehículos en reparación activa (sin fecha fin definida)
SELECT V.VIN, V.ESTADO_VEHICULO, T.NOMBRE_TALLER,
       R.FECHA_INICIO, R.FECHA_FIN
FROM REPARACION R
JOIN VEHICULO V ON V.ID_VEHICULO = R.ID_VEHICULO
JOIN TALLER T ON T.ID_TALLER = R.ID_TALLER
WHERE R.APTO_PARA_VENTA = 0;

-- ============================================================
-- PRUEBA TRG_ESTADO_VENTA_DELETE
-- Eliminar venta → vehículo vuelve a 'listo para venta'
-- ============================================================

SELECT ID_VEHICULO, ESTADO_VEHICULO FROM VEHICULO WHERE ID_VEHICULO = 4;
-- Antes: 'vendido'

DELETE FROM VENTA WHERE ID_VENTA = 1;

SELECT ID_VEHICULO, ESTADO_VEHICULO FROM VEHICULO WHERE ID_VEHICULO = 4;
-- Después: 'listo para venta'

-- ============================================================
-- PRUEBA TRG_ESTADO_VENTA_UPDATE
-- Cambiar vehículo de una venta → anterior revierte, nuevo se vende
-- ============================================================

-- Restaurar la venta
INSERT INTO VENTA VALUES (1, 1, 4, '2024-03-15', 22000.00);

SELECT ID_VEHICULO, ESTADO_VEHICULO FROM VEHICULO WHERE ID_VEHICULO IN (4, 5);
-- Ambos: 'vendido'

-- Cambiar la venta 2 (vehículo 5) para que apunte al vehículo 4
UPDATE VENTA SET ID_VEHICULO = 4 WHERE ID_VENTA = 2;

SELECT ID_VEHICULO, ESTADO_VEHICULO FROM VEHICULO WHERE ID_VEHICULO IN (4, 5);
-- Vehículo 5 (anterior) → 'listo para venta'
-- Vehículo 4 (nuevo)    → 'vendido'
